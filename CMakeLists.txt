cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DISCCENTER_VERSION LIMIT_COUNT 1)
if(NOT DISCCENTER_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-[0-9A-Za-z.-]+)?$")
  message(FATAL_ERROR "VERSION must contain a semantic version")
endif()
project(DiscCenter VERSION "${CMAKE_MATCH_1}" LANGUAGES CXX)
set(DISCCENTER_NAME "DiscCenter" CACHE STRING "Application display name")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

configure_file(cmake/identity.hpp.in generated/identity.hpp @ONLY)
add_library(disccenter_core STATIC src/core.cpp src/iso9660.cpp)
target_include_directories(disccenter_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")

include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE disccenter_core)
  add_test(NAME core_tests COMMAND core_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
  configure_file(cmake/application.rc.in generated/application.rc @ONLY)
  add_library(disccenter_windows STATIC src/platform.cpp src/optical.cpp src/iso_writer.cpp)
  target_link_libraries(disccenter_windows PUBLIC disccenter_core ole32 oleaut32 uuid shell32 shlwapi psapi bcrypt advapi32)
  add_executable(DiscCenter WIN32 src/main.cpp src/ui.cpp src/ui_actions.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/application.rc")
  target_link_libraries(DiscCenter PRIVATE disccenter_windows comctl32 comdlg32 gdi32 msimg32)
  if(MINGW)
    target_link_options(DiscCenter PRIVATE -municode -static -Wl,--major-subsystem-version,6,--minor-subsystem-version,1)
  endif()
  if(BUILD_TESTING)
    add_executable(windows_tests tests/windows_tests.cpp)
    target_link_libraries(windows_tests PRIVATE disccenter_windows)
    if(MINGW)
      target_link_options(windows_tests PRIVATE -static)
      target_link_options(core_tests PRIVATE -static)
    endif()
    add_test(NAME windows_tests COMMAND windows_tests)
  endif()
  install(TARGETS DiscCenter RUNTIME DESTINATION .)
endif()
